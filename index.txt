function shuffle(arr){
  const a = arr.slice();
  for(let i=a.length-1;i>0;i--){
    const j = Math.floor(Math.random()*(i+1));
    [a[i],a[j]]=[a[j],a[i]];
  }
  return a;
}
function freshDeck(){
  return shuffle(DECK_TEMPLATE.map((c,i)=>({...c, uid: 'c'+Date.now()+'_'+i})));
}
function makeTeam(name, code){
  const draw = freshDeck();
  const hand = draw.splice(0, HAND_SIZE);
  return {
    code, name, position:0,
    hand, draw, discard:[],
    tasksCompleted:[], log:[],
    startTime: Date.now(), finished:false, finishTime:null,
  };
}

}

function fmtTime(ms){
  const s = Math.floor(ms/1000);
  const m = String(Math.floor(s/60)).padStart(2,'0');
  const ss = String(s%60).padStart(2,'0');
  return m+':'+ss;
}

/* ---------------------------------------------------------------
   Game logic
--------------------------------------------------------------- */
function drawUpTo(n){
  while(team.hand.length < n){
    if(team.draw.length===0){
      if(team.discard.length===0) break;
      team.draw = shuffle(team.discard);
      team.discard = [];
    }
    team.hand.push(team.draw.shift());
  }
}

function playCard(uid){
  if(team.finished) return;
  const idx = team.hand.findIndex(c=>c.uid===uid);
  if(idx===-1) return;
  const card = team.hand[idx];
  team.hand.splice(idx,1);
  team.discard.push(card);

  const currentNode = NODES[team.position];
  let effectText = '';

  if(card.type==='mission'){
    if(currentNode.task && !team.tasksCompleted.includes(team.position)){
      openTaskModal(card);
      return; // 移動與抽牌會在任務完成/跳過後處理
    } else {
      effectText = '此路口沒有可觸發的任務，改為前進 1';
      advance(1);
    }
  } else if(card.type==='red'){
    effectText = '原地停等紅燈，補抽 2 張手牌';
    drawUpTo(Math.min(HAND_SIZE, team.hand.length+2));
  } else {
    effectText = card.type==='dash' ? '衝刺前進，免排隊' : '前進 ' + card.value + ' 個路口';
    advance(card.value);
  }

  logPlay(currentNode.name, card.label, effectText);
  drawUpTo(HAND_SIZE);
  finishCheck();
  renderAll();
  saveTeam();
}

function advance(steps){
  team.position = Math.min(NODES.length-1, team.position + steps);
}
function logPlay(nodeName, cardLabel, effect){
  team.log.push({ t: fmtTime(Date.now()-team.startTime), node: nodeName, card: cardLabel, effect });
}
function finishCheck(){
  if(team.position >= NODES.length-1 && !team.finished){
    team.finished = true;
    team.finishTime = Date.now();
  }
}

/* ---------------------------------------------------------------
   Dashboard（主辦方看板，讀所有隊伍共享資料）
--------------------------------------------------------------- */
async function renderDashboard(){
  const wrap = document.getElementById('dashList');
  wrap.innerHTML = '<div class="empty">讀取所有隊伍進度中…</div>';
  try{
    const listRes = await window.storage.list('team:', true);
    const keys = (listRes && listRes.keys) || [];
    if(keys.length===0){ wrap.innerHTML = '<div class="empty">目前還沒有隊伍登入</div>'; return; }
    const teams = [];
    for(const k of keys){
      try{
        const r = await window.storage.get(k, true);
        if(r) teams.push(JSON.parse(r.value));
      }catch(e){}
    }
    teams.sort((a,b)=> (b.finished - a.finished) || (b.position - a.position) || (b.tasksCompleted.length - a.tasksCompleted.length));
    wrap.innerHTML = teams.map(t=>{
      const pct = Math.round((t.position/(NODES.length-1))*100);
      return `
      <div class="team-card">
        <div class="top">
          <div class="tname">${t.name} ${t.finished ? '<span class="finished-badge">已抵達</span>' : ''}</div>
          <div class="tprog">${NODES[t.position].name}</div>
        </div>
        <div class="bar"><div style="width:${pct}%"></div></div>
        <div class="bottom">
          <span>任務 ${t.tasksCompleted.length}/${TASKS_REQUIRED}</span>
          <span>經過時間 ${fmtTime((t.finished?t.finishTime:Date.now())-t.startTime)}</span>
        </div>
      </div>`;
    }).join('');
  }catch(e){
    wrap.innerHTML = '<div class="empty">讀取失敗，請重新整理</div>';
  }
}
document.getElementById('dashRefresh').onclick = renderDashboard;
